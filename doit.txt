# 7. SPEED CALCULATION

Support both:

### GPS-provided speed

From TinyGPS++:

```cpp
gps.speed.kmph()
```

### Backend calculated speed

Using previous telemetry:

```text
previous latitude
previous longitude
previous timestamp

current latitude
current longitude
current timestamp
```

Calculate distance using Haversine.

Formula:

```text
a =
sin²((lat2-lat1)/2)
+
cos(lat1) × cos(lat2) × sin²((lon2-lon1)/2)

c = 2 × atan2(√a, √(1-a))

distance = EarthRadius × c
```

Then:

```text
speedKmh =
(distanceMeters / elapsedSeconds) × 3.6
```

Store:

```text
gpsSpeed
calculatedSpeed
effectiveSpeed
```

# 6. Telemetry Controller

Create:

```text

```

```javascript
const Device = require("../models/Device");
const Vehicle = require("../models/Vehicle");
const Telemetry = require("../models/Telemetry");

function isValidGPS(latitude, longitude) {
  if (
    typeof latitude !== "number" ||
    typeof longitude !== "number"
  ) {
    return false;
  }

  if (!Number.isFinite(latitude) || !Number.isFinite(longitude)) {
    return false;
  }

  if (latitude < -90 || latitude > 90) {
    return false;
  }

  if (longitude < -180 || longitude > 180) {
    return false;
  }

  // Treat 0,0 as invalid for this project
  if (latitude === 0 && longitude === 0) {
    return false;
  }

  return true;
}

exports.receiveTelemetry = async (req, res) => {
  try {
    const {
      deviceId,
      vehicleId,
      latitude,
      longitude,
      gpsSpeed,
      satellites,
      timestamp
    } = req.body;

    // -----------------------------------------
    // 1. Validate required fields
    // -----------------------------------------

    if (!deviceId || !vehicleId) {
      return res.status(400).json({
        success: false,
        error: "deviceId and vehicleId are required"
      });
    }

    // -----------------------------------------
    // 2. Validate GPS
    // -----------------------------------------

    if (!isValidGPS(latitude, longitude)) {
      return res.status(400).json({
        success: false,
        error: "Invalid GPS coordinates"
      });
    }

    // -----------------------------------------
    // 3. Validate device
    // -----------------------------------------

    const device = await Device.findOne({
      deviceId,
      active: true
    });

    if (!device) {
      return res.status(404).json({
        success: false,
        error: "Unknown or inactive device"
      });
    }

    // -----------------------------------------
    // 4. Verify vehicle
    // -----------------------------------------

    const vehicle = await Vehicle.findOne({
      vehicleId,
      deviceId
    });

    if (!vehicle) {
      return res.status(404).json({
        success: false,
        error: "Vehicle is not registered with this device"
      });
    }

    // -----------------------------------------
    // 5. Determine timestamp
    // -----------------------------------------

    let telemetryTime = new Date();

    if (timestamp) {
      const parsedDate = new Date(timestamp);

      if (!Number.isNaN(parsedDate.getTime())) {
        telemetryTime = parsedDate;
      }
    }

    // -----------------------------------------
    // 6. Normalize GPS speed
    // -----------------------------------------

    const safeGpsSpeed =
      typeof gpsSpeed === "number" && Number.isFinite(gpsSpeed)
        ? Math.max(0, gpsSpeed)
        : 0;

    const safeSatellites =
      typeof satellites === "number"
        ? Math.max(0, Math.floor(satellites))
        : 0;

    // -----------------------------------------
    // 7. Save telemetry
    // -----------------------------------------

    const telemetry = await Telemetry.create({
      deviceId,
      vehicleId,
      latitude,
      longitude,
      gpsSpeed: safeGpsSpeed,
      calculatedSpeed: null,
      effectiveSpeed: safeGpsSpeed,
      satellites: safeSatellites,
      timestamp: telemetryTime
    });

    // -----------------------------------------
    // 8. Update device last seen
    // -----------------------------------------

    await Device.updateOne(
      { deviceId },
      {
        $set: {
          lastSeen: new Date()
        }
      }
    );

    // -----------------------------------------
    // 9. Temporary response
    // -----------------------------------------

    return res.status(201).json({
      success: true,

      status: "RECEIVED",

      deviceId,

      vehicleId,

      location: {
        latitude,
        longitude
      },

      speed: {
        gpsSpeed: safeGpsSpeed,
        effectiveSpeed: safeGpsSpeed
      },

      satellites: safeSatellites,

      timestamp: telemetryTime,

      telemetryId: telemetry._id
    });
  } catch (error) {
    console.error("Telemetry error:", error);

    return res.status(500).json({
      success: false,
      error: "Internal server error"
    });
  }
};
```

```cpp
// =====================================================
// READ GPS
// =====================================================

void readGPS() {

  while (gpsSerial.available() > 0) {

    char c = gpsSerial.read();

    gps.encode(c);
  }

// =====================================================
// CONNECT WIFI
// =====================================================

bool connectWiFi() {

  if (WiFi.status() == WL_CONNECTED) {
    return true;
  }

  Serial.println();
  Serial.println("Connecting to WiFi...");

  Serial.print("SSID: ");
  Serial.println(ssid);

  WiFi.begin(
    ssid,
    password
  );

  int attempts = 0;

  while (
    WiFi.status() != WL_CONNECTED &&
    attempts < 30
  ) {

    delay(500);

    Serial.print(".");

    attempts++;
  }

  if (WiFi.status() == WL_CONNECTED) {

    Serial.println();
    Serial.println("WiFi connected!");

    Serial.print("ESP32 IP: ");
    Serial.println(WiFi.localIP());

    return true;
  }

  Serial.println();
  Serial.println("WiFi connection failed.");

  return false;
}

// =====================================================
// SEND TELEMETRY
// =====================================================

void sendTelemetry(
    double latitude,
    double longitude,
    double speed,
    int satellites
) {

  // ---------------------------------------------
  // Check WiFi
  // ---------------------------------------------

  if (!connectWiFi()) {

    Serial.println(
      "Cannot send telemetry: WiFi unavailable."
    );

    return;
  }

  // ---------------------------------------------
  // Create timestamp
  // ---------------------------------------------

  // For this first implementation, server time
  // is authoritative.
  //
  // We still send millis() for debugging.

  unsigned long deviceMillis = millis();

  // ---------------------------------------------
  // JSON
  // ---------------------------------------------

  String jsonData = "{";

  jsonData += "\"deviceId\":\"";
  jsonData += deviceId;
  jsonData += "\",";

  jsonData += "\"vehicleId\":\"";
  jsonData += vehicleId;
  jsonData += "\",";

  jsonData += "\"latitude\":";
  jsonData += String(latitude, 6);
  jsonData += ",";

  jsonData += "\"longitude\":";
  jsonData += String(longitude, 6);
  jsonData += ",";

  jsonData += "\"gpsSpeed\":";
  jsonData += String(speed, 2);
  jsonData += ",";

  jsonData += "\"satellites\":";
  jsonData += String(satellites);
  jsonData += ",";

  jsonData += "\"deviceMillis\":";
  jsonData += String(deviceMillis);

  jsonData += "}";

  // ---------------------------------------------
  // Display
  // ---------------------------------------------

  Serial.println();
  Serial.println("=================================");
  Serial.println("SENDING TELEMETRY");
  Serial.println("=================================");

  Serial.println(jsonData);

  // ---------------------------------------------
  // HTTP
  // ---------------------------------------------

  HTTPClient http;

  http.setTimeout(10000);

  Serial.print("Connecting to: ");
  Serial.println(serverUrl);

  if (!http.begin(serverUrl)) {

    Serial.println(
      "HTTPClient initialization failed."
    );

    return;
  }

  http.addHeader(
    "Content-Type",
    "application/json"
  );

  int responseCode =

  Serial.print(
    "HTTP Response Code: "
  );

  Serial.println(responseCode);

  // ---------------------------------------------
  // Response
  // ---------------------------------------------

  if (responseCode > 0) {

    Serial.println("SERVER RESPONSE:");
    Serial.println(response);

    Serial.println("HTTP REQUEST FAILED");

    Serial.print(
      "Error: "
    );

    Serial.println(
      http.errorToString(responseCode)
    );
  }

  http.end();
}
```

```text

                       │
                       ▼
              ┌─────────────────┐
```
